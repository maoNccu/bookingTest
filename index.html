// 動態載入時段卡片按鈕 function loadTimeslots() { const listEl = document.getElementById('timeslotList'); fetch(GAS_API_URL) .then(response => response.json()) .then(res => { if (res.result === 'success') { listEl.innerHTML = ''; if (res.data.length === 0) { listEl.innerHTML = '

目前無可用時段。

'; return; } res.data.forEach(item => { const card = document.createElement('div'); card.className = `timeslot-card ${item.isFull ? 'disabled' : ''}`; card.innerHTML = `
⏰ ${item.timeslot} 🎯 ${item.topic}
${item.isFull ? '已額滿' : `剩 ${item.remaining} 名`}
`; // 如果未額滿，綁定點擊事件 if (!item.isFull) { card.addEventListener('click', function(e) { e.preventDefault(); // 1. 移除其他卡片的 selected 樣式 const allCards = document.querySelectorAll('.timeslot-card'); allCards.forEach(c => c.classList.remove('selected')); // 2. 當前卡片加上 selected 樣式 card.classList.add('selected'); // 3. 將選取的時段寫入隱藏欄位 const hiddenInput = document.getElementById('selectedTimeslot'); hiddenInput.value = item.timeslot; console.log('已選取時段：', hiddenInput.value); // 可在瀏覽器主控台 (F12) 查看 }); } listEl.appendChild(card); }); } else { listEl.innerHTML = '

載入時段失敗：' + (res.message || '') + '

'; } }) .catch(error => { console.error('Error:', error); listEl.innerHTML = '

網路連線錯誤，請重試。

'; }); }
